# 第一部分：产品设计文档
## 一、系统总述
装备系统是寻仙核心成长线之一，与人物系统、副本系统、野怪系统深度耦合。玩家通过副本掉落、野怪掉落、打造三条路径获取装备，通过穿戴装备提升精/神/气三围属性及附加战斗属性。装备系统同时承载了神位继承的关键入口——神话碎片合成完整神话后开启神位继承资格，将装备系统与PVP对抗（大道争锋）串联。
本系统覆盖以下核心模块：装备槽位结构、品质体系、基础加成、套装效果、装备等级与升级、附加属性、武器类型与攻击形状、耐久与修理、打造系统、掉落系统、神话碎片与神位继承、腰带特殊功能、仓库系统、装备外观、交易系统、穿戴境界限制。

## 四、基础加成系统
装备基础加成只增加精/神/气三种属性。三大类装备分别对应不同属性：

| \u88c5\u5907\u5927\u7c7b | \u52a0\u6210\u5c5e\u6027 | \u51e1\u54c18\u4ef6/4\u4ef6/2\u4ef6\u603b\u548c\uff081\u7ea7\uff09 | \u5206\u914d\u65b9\u5f0f |

| \u6b66\u5668\uff08\u4e3b+\u526f\uff09 | \u6c14 | 1\uff081\u4e3b1\u526f\uff09 | \u72ec\u7acb\u8ba1\u7b97\u6c42\u548c |

### 4.1 甲类加成规则
8件甲独立计算品质加成，每件按自身品质×（凡品基准1精/8件）
每件装备生成时随机分配属性值，8件总和=品质倍率×1精
不同部位可不同品质，混穿独立求和
示例：凡品8件甲总和=1精（随机分配到8件上），珍品8件甲总和=2精，先天灵宝8件甲总和=12精。
### 4.2 首饰加成规则
4件首饰独立计算品质加成，每件按自身品质×（凡品基准1神/4件）
每件装备生成时随机分配属性值，4件总和=品质倍率×1神
不同部位可不同品质，混穿独立求和
### 4.3 武器加成规则
主武器和副武器独立计算品质加成
副武器按实际装备数线性插值（见2.1节）
主武器与副武器可不同品质，各自独立计算后求和
示例：主武器凡品+1副凡品=1气；主武器仙宝+5副仙品=5×5=25气；主武器凡品+5副珍品=1×1+5×2/3≈4.33气。

## 六、套装效果
### 6.1 套装触发条件
穿戴甲8件 + 首饰4件全部同品质，触发套装效果
武器不参与套装判定
穿齐套装那一刻系统随机生成0~80%的浮动值，锁定
更换任意一件装备则套装解除，重新穿齐后重新计算浮动值
### 6.2 套装加成计算
套装基础值 = 品质倍率 × 1（甲基础1精 + 首饰基础1神的合并基数）
套装浮动加成 = 套装基础值 × 随机浮动值(0~80%)
浮动加成随机分配到精/神/气其中一个属性上。
示例：凡品全套（8甲+4首饰同凡品），基础值=1，浮动随机到50%，则额外加成=0.5，随机分配到精/神/气其中一个。
示例：先天灵宝全套，基础值=12，浮动随机到80%，则额外加成=9.6，随机分配到精/神/气其中一个。

## 九、装备耐久与修理
### 9.1 耐久系统
每件装备耐久上限：300点

## 四、装备生成系统
装备生成在掉落/打造时触发，服务端生成装备实例。
// Go伪代码
type EquipmentInstance struct {
    ID           int64
    TemplateID   int
    Quality      Quality  // 凡品/珍品/灵宝/仙宝/神话/道宝/先天灵宝/功德灵宝
    SlotType     SlotType // 头甲/面甲/.../主武器/副武器
    WeaponType   WeaponType // 刀/剑/枪/...（仅武器）
    Level        int      // 1~10
    Durability   int      // 300
    BaseAttrDist map[AttrType]float64 // 精气神分配
    ExtraAttrs   []ExtraAttr
    IsEquipped   bool
    OwnerID      int64
}

func GenerateEquipment(quality Quality, slotType SlotType, level int) *EquipmentInstance {
    eq := &EquipmentInstance{
        Quality:    quality,
        SlotType:   slotType,
        Level:      level,
        Durability: 300,
    }
    // 1. 基础属性随机分配
    eq.BaseAttrDist = randomDistributeBase(quality, slotType)
    // 2. 附加属性生成
    eq.ExtraAttrs = generateExtraAttrs(quality, level)
    return eq
}

func randomDistributeBase(quality Quality, slotType SlotType) map[AttrType]float64 {
    totalBase := qualityMultiplier(quality) * slotBaseValue(slotType) // 如凡品甲=1*1=1精
    // 全随机分配到各件（此处生成单件时随机其分得的值）
    // 甲类8件总和=totalBase，每件随机0~totalBase，最后归一化
    return map[AttrType]float64{
        slotAttrType(slotType): rand.Float64() * totalBase, // 简化：单件随机分得
    }
}
## 五、加成计算系统
### 5.1 总加成计算
// Go伪代码
func CalcTotalBonus(playerID int64) PlayerBonus {
    equipped := equipmentService.GetEquipped(playerID)
    var total PlayerBonus
    for _, eq := range equipped {
        // 基础加成 = 品质倍率 × 基数 × 等级系数
        levelFactor := 1.0 + float64(eq.Level-1)/9.0
        baseBonus := qualityMultiplier(eq.Quality) * slotBaseValue(eq.SlotType) * levelFactor
        total.Add(eq.SlotType.AttrType(), baseBonus)

        // 附加属性加成
        for _, attr := range eq.ExtraAttrs {
            attrValue := attr.Value * levelFactor // 附加属性也随等级提升
            total.AddExtra(attr.Type, attrValue)
        }
    }
    // 套装效果
    total.AddSetBonus(calcSetBonus(equipped))
    return total
}

// Go伪代码
func CalcSetBonus(equipped []EquipmentInstance) SetBonus {
    // 检查8甲+4首饰是否全同品质
    armorQualities := getQualitiesByCategory(equipped, CATEGORY_ARMOR)
    accQualities := getQualitiesByCategory(equipped, CATEGORY_ACCESSORY)
    if !allSame(armorQualities) || !allSame(accQualities) {
        return SetBonus{} // 无套装
    }
    if armorQualities[0] != accQualities[0] {
        return SetBonus{} // 甲和首饰品质不同
    }
    quality := armorQualities[0]
    // 查询或生成浮动值
    setBonus := setBonusService.GetOrCreate(playerID, quality)
    if !setBonus.IsActive {
        // 穿齐时生成浮动值
        setBonus.FloatValue = rand.Float64() * 0.8 // 0~80%
        setBonus.FloatAttr = randomAttrType()       // 随机精/神/气
        setBonus.IsActive = true
        setBonusService.Save(setBonus)
    }
    baseValue := qualityMultiplier(quality) * 1.0
    bonusValue := baseValue * setBonus.FloatValue
    return SetBonus{Attr: setBonus.FloatAttr, Value: bonusValue}
}

// Go伪代码
func SubWeaponBaseValue(count int) float64 {
    // 凡品基数：1副=1, 5副=3, 8副=5
    if count <= 1 {
        return 1.0
    }
    if count <= 5 {
        return 1.0 + float64(count-1)*(3.0-1.0)/(5.0-1.0)
    }
    if count <= 8 {
        return 3.0 + float64(count-5)*(5.0-3.0)/(8.0-5.0)
    }
    return 5.0
}

// Go伪代码
func Craft(playerID int64, recipeID int, materials []Item) (*CraftResult, error) {
    recipe := configService.GetRecipe(recipeID)
    craftLevel := craftService.GetLevel(playerID)

    // 1. 检查打造品级是否满足
    if craftLevel < recipe.RequiredLevel {
        return nil, ErrCraftLevelTooLow
    }
    // 2. 检查材料
    if !checkMaterials(materials, recipe.Materials) {
        return nil, ErrMaterialsNotEnough
    }
    // 3. 计算成功率
    luck := playerService.GetLuck(playerID)
    successRate := recipe.SuccessRate + luck*0.1
    if successRate > 1.0 {
        successRate = 1.0
    }
    // 4. 扣除材料
    inventoryService.RemoveItems(playerID, materials)
    inventoryService.RemoveGold(playerID, recipe.GoldCost)
    // 5. 判定
    if rand.Float64() > successRate {
        // 失败：材料已消耗
        return &CraftResult{Success: false}, nil
    }
    // 6. 成功：生成装备
    level := rand.Intn(10) + 1 // 随机1~10级
    eq := GenerateEquipment(recipe.Quality, recipe.SlotType, level)
    eq.OwnerID = playerID
    equipmentService.Save(eq)
    // 7. 更新打造计数
    craftService.IncrementSuccessCount(playerID, recipe.Quality)
    craftService.TryLevelUp(playerID)
    return &CraftResult{Success: true, Equipment: eq}, nil
}

    if successRate > 1.0 {
        successRate = 1.0
    }

        }
    }

}

func OnDurabilityConsume(equipmentID int64, amount int) error {

    if eq.Durability <= 10 && eq.Durability > 0 {

}

    return nil
}

// Go伪代码
var upgradeRates = map[int]float64{
    1: 1.0, 2: 0.95, 3: 0.90, 4: 0.80, 5: 0.70,
    6: 0.60, 7: 0.55, 8: 0.52, 9: 0.50,
}

var upgradeFailures = map[int]string{
    2: "DOWNGRADE", 3: "DOWNGRADE", 4: "DOWNGRADE",
    5: "HALF_DOWN_HALF_DESTROY", 6: "HALF_DOWN_HALF_DESTROY",
    7: "DESTROY", 8: "DESTROY", 9: "DESTROY",
}

func Upgrade(playerID int64, equipmentID int64, materials []Item) (*UpgradeResult, error) {
    eq := equipmentService.Get(equipmentID)
    if eq.Level >= 10 {
        return nil, ErrMaxLevel
    }
    // 扣除材料和钱
    inventoryService.RemoveItems(playerID, materials)
    inventoryService.RemoveGold(playerID, upgradeCost(eq))
    // 计算成功率
    luck := playerService.GetLuck(playerID)
    rate := upgradeRates[eq.Level] + luck*0.1
    if rate > 1.0 {
        rate = 1.0
    }
    if rand.Float64() <= rate {
        eq.Level++
        equipmentService.Save(eq)
        return &UpgradeResult{Success: true, NewLevel: eq.Level}, nil
    }
    // 失败处理
    failType := upgradeFailures[eq.Level]
    switch failType {
    case "DOWNGRADE":
        if eq.Level > 1 {
            eq.Level--
        }
    case "HALF_DOWN_HALF_DESTROY":
        if rand.Float64() < 0.5 {
            if eq.Level > 1 { eq.Level-- }
        } else {
            equipmentService.Delete(equipmentID)
            return &UpgradeResult{Success: false, Destroyed: true}, nil
        }
    case "DESTROY":
        equipmentService.Delete(equipmentID)
        return &UpgradeResult{Success: false, Destroyed: true}, nil
    }
    equipmentService.Save(eq)
    return &UpgradeResult{Success: false, Downgraded: true, NewLevel: eq.Level}, nil
}

    ID          int64
    SellerID    int64
    BuyerID     int64
    EquipmentID int64
    Price       int64
    Status      string // PENDING/COMPLETED/CANCELLED
    CreatedAt   time.Time
}

func CreateTrade(sellerID int64, equipmentID int64, price int64) (*TradeOrder, error) {
    eq := equipmentService.Get(equipmentID)
    if eq.OwnerID != sellerID {
        return nil, ErrNotOwner
    }
    // 装备不绑定，可直接交易
    order := &TradeOrder{
        SellerID: sellerID, EquipmentID: equipmentID,
        Price: price, Status: "PENDING",
    }
    tradeService.Save(order)
    return order, nil
}

func ExecuteTrade(orderID int64, buyerID int64) error {
    order := tradeService.Get(orderID)
    // 扣除买家金币
    inventoryService.RemoveGold(buyerID, order.Price)
    // 转移装备
    eq := equipmentService.Get(order.EquipmentID)
    eq.OwnerID = buyerID
    if eq.IsEquipped {
        // 如果装备正在穿戴，先卸下
        equipmentService.Unequip(order.SellerID, eq.ID)
    }
    equipmentService.Save(eq)
    // 增加卖家金币
    inventoryService.AddGold(order.SellerID, order.Price)
    order.Status = "COMPLETED"
    tradeService.Save(order)
    return nil
}
