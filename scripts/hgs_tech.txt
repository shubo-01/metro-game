[P80][] // Go 伪代码
[P81][] func CalcProbs(luck float64) map[int]float64 {
[P82][] wkProb := 0.01 * (1 + luck)  // 孙悟空
[P83][] mwProb := 0.09 * (1 + luck)  // 魔王
[P84][] extra  := (wkProb - 0.01) + (mwProb - 0.09)
[P85][] poolSum := 0.30 + 0.35 + 0.10 + 0.15  // 0.90
[P86][] // 四角色按比例扣除
[P87][] probs[3] = 0.30 - extra * (0.30 / poolSum)  // 手下
[P88][] probs[4] = 0.35 - extra * (0.35 / poolSum)  // 猴子
[P89][] probs[5] = 0.10 - extra * (0.10 / poolSum)  // 老猴
[P90][] probs[6] = 0.15 - extra * (0.15 / poolSum)  // 草木石头
[P91][] return probs
[P92][] }
